\section{Ordinals}\label{sec:ordinals}

\paragraph{Hereditarily transitive sets}

\begin{definition}\label{def:hereditarily_transitive_set}\mimprovised
  We say that a \hyperref[def:transitive_set]{transitive set} is \term{hereditarily transitive} if its members are also transitive sets.
\end{definition}
\begin{comments}
  \item As we will see in \cref{thm:def:hereditarily_transitive_set/recursion}, this definition applies recursively, which justifies the name.

  \item With the \hyperref[def:zfc/foundation]{axiom of foundation}, the hereditarily transitive sets are precisely the \hyperref[def:ordinal]{ordinals}. Hence, a separate notion of a hereditarily transitive set is not established.

  We extract this into a separate definition to study properties that only depend on transitivity.

  \item We give some important examples related to transitive sets in \cref{ex:set_encoding_of_natural_numbers}.
\end{comments}

\begin{proposition}\label{thm:def:hereditarily_transitive_set}
  \hyperref[def:hereditarily_transitive_set]{Hereditarily transitive sets} have the following basic properties:
  \begin{thmenum}
    \thmitem{thm:def:hereditarily_transitive_set/recursion} All members of a hereditarily transitive set are also hereditarily transitive.

    \thmitem{thm:def:hereditarily_transitive_set/successor} The \hyperref[def:ordinal_successor]{successor} of a \hyperref[def:hereditarily_transitive_set]{hereditarily transitive set} is also hereditarily transitive.

    \thmitem{thm:def:hereditarily_transitive_set/relation} A transitive set \( A \) is hereditarily transitive if and only if the \hyperref[def:relation/restriction]{restriction} of set membership to \( A \) is a \hyperref[def:binary_relation/transitive]{transitive relation}.

    \thmitem{thm:def:hereditarily_transitive_set/union} Hereditarily transitive sets are closed under \hyperref[def:set_union/unary]{unary unions}.

    \thmitem{thm:def:hereditarily_transitive_set/intersection} Hereditarily transitive sets are closed under \hyperref[def:set_intersection/unary]{unary intersection}.
  \end{thmenum}
\end{proposition}
\begin{comments}
  \item \Cref{thm:def:ordinal} specializes some of these properties for \hyperref[def:ordinal]{ordinals}.
\end{comments}
\begin{proof}
  \SubProofOf{thm:def:hereditarily_transitive_set/recursion} If \( A \) is hereditarily transitive and \( x \) belongs to \( A \), then \( x \) is by definition transitive. If \( y \) belongs to \( x \), it also belongs to \( A \) since \( A \) is transitive. Then \( y \) is transitive as a member of \( A \). Generalizing on \( y \), we conclude that \( x \) is hereditarily transitive.

  \SubProofOf{thm:def:hereditarily_transitive_set/successor} Suppose that \( A \) is hereditarily transitive. \Cref{thm:successor_of_transitive_set} implies that \( S(A) \) is transitive, so we must show that its members are also transitive.

  We must show that every member \( x \) of \( S(A) = A \cup \set{ A } \) is transitive. Either \( x \) belongs to \( A \) or equals \( A \) --- in both cases it is by assumption transitive.

  \SubProofOf{thm:def:hereditarily_transitive_set/relation} Suppose that \( A \) is a transitive set. Let \( x \), \( y \) and \( z \) be members of \( A \) such that
  \begin{equation*}
    z \in y \in x.
  \end{equation*}

  Then \( z \) belongs to \( x \) for any \( y \) and \( z \) if and only if \( x \) is a transitive set.

  \SubProofOf{thm:def:hereditarily_transitive_set/union} Let \( \mscrA \) be a collection of hereditarily transitive sets.

  If \( x \) belongs to the union \( \bigcup \mscrA \), there exists some \( A \) in \( \mscrA \) such that \( x \) belongs to \( A \). Since \( A \) is by assumption transitive, \( x \) is a subset of \( A \), and hence of \( \bigcup \mscrA \).

  Furthermore, by \cref{thm:def:hereditarily_transitive_set/recursion}, \( A \) is hereditarily transitive, and thus \( x \) is itself transitive.

  It follows that \( \bigcup \mscrA \) is hereditarily transitive.

  \SubProofOf{thm:def:hereditarily_transitive_set/intersection} Let \( \mscrA \) be a nonempty collection of hereditarily transitive sets.

  If \( x \) belongs to the intersection \( \bigcap \mscrA \), then \( x \) belongs to every set \( A \) in \( \mscrA \). Then \( x \) is a subset of every \( A \) in \( \mscrA \), and hence of \( \bigcap \mscrA \).

  Furthermore, by \cref{thm:def:hereditarily_transitive_set/recursion}, \( A \) is hereditarily transitive, and thus \( x \) is itself transitive.

  It follows that \( \bigcap \mscrA \) is hereditarily transitive.
\end{proof}

\begin{lemma}\label{thm:well_founded_hereditarily_transitive_set}\mcite{MathSE:transitive_set_of_transitive_sets}
  If, in a \hyperref[def:hereditarily_transitive_set]{hereditarily transitive set}, membership is \hyperref[def:well_founded_relation]{well-founded}, it is automatically a \hyperref[def:totally_ordered_set]{total order}.
\end{lemma}
\begin{comments}
  \item Assuming the \hyperref[def:zfc/foundation]{axiom of foundation}, all sets are well-founded, so the hereditarily finite sets are automatically well-ordered, i.e. ordinals.
\end{comments}
\begin{proof}
  Suppose that \( A \) is hereditarily transitive and that the restriction of \( {\in} \) to \( A \) is well-founded. Then, in particular, \( {\in} \) is irreflexive. Furthermore, \cref{thm:def:hereditarily_transitive_set/relation} implies that it is transitive relation, so it is a strict partial order on \( A \).

  It remains to show that \( {\in} \) is \hyperref[def:totally_ordered_set/trichotomic]{trichotomic}, i.e. that all members of \( A \) are comparable. Aiming at a contradiction, suppose that \( A \) has incomparable elements.

  Consider the sets
  \begin{align*}
    B    &\coloneqq \set{ x \in A \given \text{some } y \in A \T{is incomparable with} x }, \\
    C(b) &\coloneqq \set{ y \in A \given y \T{is incomparable with} b }.
  \end{align*}

  By assumption, \( B \) is nonempty, so, by well-foundedness, it has a minimal member \( b \). Then \( C(b) \) also has a minimal member \( c \).

  Minimality of \( c \) in \( C(b) \) means that every member \( z \) of \( c \) is comparable with \( b \). We have three possibilities:
  \begin{itemize}
    \item If \( b = z \), then \( b \in c \), which contradicts the choice of \( c \).
    \item If \( b \in z \), then \( b \in c \) by the transitivity of \( c \), which also contradicts the choice of \( c \).
    \item It remains for \( z \in b \) to hold.
  \end{itemize}

  Since \( z \) was arbitrary, we conclude that \( c \) is a subset of \( b \).

  Minimality of \( b \) in \( B \) means that all members of \( b \) are comparable with all of \( A \). In particular, every member \( y \) of \( b \) is comparable with \( c \). Again, we have three possibilities:
  \begin{itemize}
    \item If \( c = y \), then obviously \( c \in b \), which contradicts the choice of \( c \).
    \item If \( c \in y \), then \( c \in b \) by the transitivity of \( b \), which also contradicts the choice of \( c \).
    \item It remains for \( y \in c \) to hold.
  \end{itemize}

  Since \( y \) was arbitrary, we conclude that \( b \) is a subset of \( c \). Then \( b = c \), which contradicts their incomparability.

  Therefore, we conclude that all members of \( A \) are comparable.
\end{proof}

\begin{corollary}\label{thm:hereditarily_transitive_set_with_foundation}
  Assuming the \hyperref[def:zfc/foundation]{axiom of foundation}, a \hyperref[def:transitive_set]{transitive} set \( A \) is \hyperref[def:hereditarily_transitive_set]{hereditarily transitive} if and only if set membership \hyperref[def:totally_ordered_set]{totally orders} \( A \).
\end{corollary}
\begin{comments}
  \item We take the latter as a defining property for ordinals in \cref{def:ordinal}.
\end{comments}
\begin{proof}
  \SufficiencySubProof Suppose that \( A \) is hereditarily transitive. By the axiom of foundation, set membership in \( A \) is \hyperref[def:well_founded_relation]{well-founded} and, as a consequence of \cref{thm:well_founded_hereditarily_transitive_set}, it is also a total order.

  \NecessitySubProof Suppose that \( A \) is transitive and that set membership is a total order on \( A \). In particular, membership is transitive, and \cref{thm:def:hereditarily_transitive_set/relation} implies that \( A \) is hereditarily transitive.
\end{proof}

\begin{definition}\label{def:set_transitive_closure}\mcite[286]{Mendelson2015Logic}
  We define the \term{transitive closure} of a set \( A \) as
  \begin{equation*}
    \cl^T(A) \coloneqq \bigcup_{n=0}^\infty {\bigcup}^n A,
  \end{equation*}
  where \( {\bigcup}^n \) is the \( n \)-fold composition of the \hyperref[def:set_union/unary]{unary set union}.
\end{definition}
\begin{comments}
  \item Formally, we must take the unary union of the \hyperref[def:set_valued_function/image]{image} of the following \hyperref[def:sequence]{sequence} defined via \fullref{thm:omega_recursion}:
  \begin{equation*}
    T_n \coloneqq \begin{cases}
      A,               &n = 0, \\
      \bigcup T_{n-1}, &n > 0.
    \end{cases}
  \end{equation*}

  \item \Cref{thm:set_transitive_closure} implies that the transitive closure satisfies the conditions for a \hyperref[def:abstract_closure_operator]{closure operator} on the \hyperref[def:lattice_of_sets]{lattice of all sets}.

  We can regard it as a metatheoretic operator on a model of set theory.

  \item This concept is fundamentally different from that of the transitive closure of a relation defined in \cref{def:relation_closures/transitive}.
\end{comments}

\begin{proposition}\label{thm:set_transitive_closure}
  The \hyperref[def:set_transitive_closure]{transitive closure} \( \cl^T(A) \) of the set \( A \) is the smallest \hyperref[def:transitive_set]{transitive set} containing \( A \).

  More formally, \( \cl^T(A) \) is transitive and every transitive set that contains \( A \) also contains \( \cl^T(A) \).
\end{proposition}
\begin{proof}
  \SubProof{Proof that \( \cl^T(A) \) is transitive} Fix some member \( x \) of \( \cl^T(A) \). Then \( x \) belongs to \( {\bigcup}^n A \) for some \( n \). \Cref{thm:unary_set_union_inequality} implies that \( x \) is a subset of \( {\bigcup}^{n+1} A \), and thus of \( \cl^T(A) \).

  It follows that \( \cl^T(A) \) is transitive.

  \SubProof{Proof that \( \cl^T(A) \) is minimal} Clearly \( A = {\bigcup}^0 A \) is a subset of \( \cl^T(A) \).

  Let \( B \) be any transitive set containing \( A \). We will use \fullref{thm:omega_induction} on \( n \) in \( \omega \) to show that \( \bigcup^n A \) is a subset of \( B \).
  \begin{itemize}
    \item The case \( n = 0 \) is obvious since \( B \) is assumed to be a superset of \( A = {\bigcup}^0 A \).

    \item If \( {\bigcup}^n A \) is a subset of \( B \), we must show that \( {\bigcup}^{n+1} A \) also is.

    Fix some \( x \) from \( {\bigcup}^{n+1} A \). By definition of union, \( x \) belongs to some member \( A' \) of \( {\bigcup}^n A \).

    We have assumed that \( {\bigcup}^n A \) is a subset of \( B \), so \( A' \) belongs to \( B \). Since \( B \) is transitive, the member \( x \) of \( A' \) also belongs to \( B \).

    Therefore, \( {\bigcup}^{n+1} A \) is a subset of \( B \).
  \end{itemize}

  Thus, the union
  \begin{equation*}
    \cl^T(A) = \set*{ {\bigcup}^n A \given* n \in \omega }
  \end{equation*}
  is also a subset of \( B \).
\end{proof}

\begin{corollary}\label{thm:closure_of_collection_of_hereditarily_transitive_sets}
  The \hyperref[def:set_transitive_closure]{transitive closure} of a collection of \hyperref[def:hereditarily_transitive_set]{hereditarily transitive sets} is itself hereditarily transitive.
\end{corollary}
\begin{comments}
  \item \Cref{thm:def:ordinal/closure} restates this proposition for \hyperref[def:ordinal]{ordinals}.
\end{comments}
\begin{proof}
  Let \( A \) be a collection of hereditarily transitive sets. \Cref{thm:set_transitive_closure} implies that the closure \( \cl^T(A) \) is transitive.

  We will use \fullref{thm:omega_induction} on \( n \) to show that the members of \( {\bigcup}^n A \) are hereditarily transitive.
  \begin{itemize}
    \item This is vacuous for \( n = 0 \) since, by \cref{thm:def:hereditarily_transitive_set/recursion}, the members of \( A = {\bigcup}^0 A \) are transitive.

    \item Suppose that the members of \( {\bigcup}^n A \) are hereditarily transitive.

    If \( x \) belongs to \( {\bigcup}^{n+1} A \), it belongs to some set \( A' \) in \( {\bigcup}^n A \). The latter is hereditarily transitive by assumption, so, by \cref{thm:def:hereditarily_transitive_set/recursion}, \( x \) is also hereditarily transitive.
  \end{itemize}
\end{proof}

\paragraph{Ordinals}

\begin{definition}\label{def:ordinal}
  We say that the set \( \alpha \) is an \term[ru=ординал (\cite[\S 1.5.3]{КусраевКутателадзе2005БулевозначныйАнализ}), en=ordinal (\cite[def. I.7.3]{Kunen2013SetTheory})]{ordinal} if it satisfies any of the following equivalent conditions:
  \begin{thmenum}
    \thmitem{def:ordinal/neumann}\mcite[def. I.7.3]{Kunen2013SetTheory} \( \alpha \) is a \hyperref[def:transitive_set]{transitive set} that is \hyperref[def:well_ordered_set]{well-ordered} by set membership.

    This definition is due to \textcite[203]{VonNeumann1923TransfinitenZahlen}. If disambiguation is necessary, we call any transitive well-ordered set a \term{von Neumann ordinal}.

    \thmitem{def:ordinal/hereditarily_transitive}\mimprovised \( \alpha \) is a \hyperref[def:hereditarily_transitive_set]{hereditarily transitive set} in which set membership is \hyperref[def:well_founded_relation]{well-founded}.

    Based on this definition, we formally introduce, in accordance with \cref{def:set_theory_definition/predicate}, a unary \hyperref[def:fol_signature/predicate]{predicate symbol}:
    \begin{align*}
      \synpred{ordinal}(\syna) &\coloneqq \synpred{transitive}(\syna) \synwedge \qforall \synb \synpred{transitive}(\synb) \synwedge \synpred{well-founded}(\syna),
    \end{align*}

    We have used the \( \synpred{transitive} \) predicate from \cref{def:transitive_set} and the \( \synpred{well-founded} \) predicate from \cref{def:zfc/foundation}.
  \end{thmenum}
\end{definition}
\begin{comments}
  \item Well-foundedness is automatic in the presence of the \hyperref[def:zfc/foundation]{axiom of foundation}, so, by \cref{thm:hereditarily_transitive_set_with_foundation}, the ordinals are precisely the hereditarily transitive sets. In fact, ordinals are defined as hereditarily transitive sets in some books like \cite[246]{Shoenfield1967MathematicalLogic} and \cite[86]{ШеньВерещагин2020НачалаТеорииМножеств}.

  Still, at least in this section, we prefer to be more explicit.

  \item The ordinals themselves are \hyperref[def:well_ordered_set]{well-ordered} by membership --- see \cref{def:ordinal_ordering}.

  \item Within the metalanguage, we denote abstract ordinals via initial Greek letters like \( \alpha \), \( \beta \), \( \gamma \), in accordance with \cref{rem:mathematical_logic_conventions/greek_alphabet}.

  \item An early history of the concept of ordinals, starting from Cantor's notion of order types, is given in a preface to the English translation \cite{VanHeijenoort2002LogicCollection:TransfiniteNumbers} of von Neumann's paper.
\end{comments}

\begin{proposition}\label{thm:def:ordinal}
  \hyperref[def:ordinal]{Ordinals} have the following basic properties:
  \begin{thmenum}
    \thmitem{thm:def:ordinal/recursion} All members of a ordinal are also ordinals.

    \thmitem{thm:def:ordinal/successor} The \hyperref[def:ordinal_successor]{successor} of a \hyperref[def:ordinal]{ordinal} is also an ordinal.

    \thmitem{thm:def:ordinal/union} Ordinals are closed under \hyperref[def:set_union/unary]{unary unions}.

    \thmitem{thm:def:ordinal/intersection} Ordinals are closed under \hyperref[def:set_intersection/unary]{unary intersection}.

    \thmitem{thm:def:ordinal/transitive_set} A \hyperref[def:transitive_set]{transitive set} is an ordinal if and only if its members are ordinals.

    \thmitem{thm:def:ordinal/closure} The \hyperref[def:set_transitive_closure]{transitive closure} of a collection of ordinals is itself an ordinal.
  \end{thmenum}
\end{proposition}
\begin{proof}
  \SubProofOf{thm:def:ordinal/recursion} Fix an ordinal \( \alpha \) and let \( A \) be any member of \( \alpha \). By \cref{thm:def:hereditarily_transitive_set/recursion}, \( A \) is hereditarily transitive.

  We will show that membership in \( A \) is well-founded. Fix a nonempty subset \( B \) of \( A \). Since \( \alpha \) is transitive, \( B \) is a subset of \( \alpha \), and hence has a minimal element \( m \) in \( \alpha \). Thus, there exists no \( x \) in \( \alpha \) such that \( x \in m \).

  Since \( A \) belongs to \( \alpha \) and since the latter is transitive, \( A \) is a subset of \( \alpha \), and hence there exists no \( x \) in \( A \) such that \( x \in m \); that is, \( m \) is minimal in \( A \).

  Generalizing on \( m \) and \( B \), we conclude that membership in \( A \) is well-founded. Therefore, \( A \) is an ordinal.

  \SubProofOf{thm:def:ordinal/successor} Fix an ordinal \( \alpha \). \Cref{thm:def:hereditarily_transitive_set/successor} implies that its successor \( S(\alpha) = \alpha \cup \set{ \alpha } \) is also an hereditarily transfinite set.

  Furthermore, \( S(\alpha) \) is well-founded since every nonempty subset \( A \) has a minimum --- either \( A = \set{ \alpha } \) and the minimum is \( \alpha \), or otherwise it is the minimum of \( A \cap \alpha \).

  Therefore, \( S(\alpha) \) is an ordinal.

  \SubProofOf{thm:def:ordinal/union} Let \( A \) be a collection of ordinals. \Cref{thm:def:hereditarily_transitive_set/union} implies that its union \( \bigcup A \) is hereditarily transitive.

  Furthermore, \cref{thm:def:ordinal/recursion} implies that every member of \( \bigcup A \) is an ordinal since it belongs to some ordinal in \( A \).

  We must show that \( \bigcup A \) is well-founded. Indeed, if it contains an infinitely descending chain
  \begin{equation*}
    \cdots \in \alpha_n \in \cdots \in \alpha_2 \in \alpha_1,
  \end{equation*}
  this contradicts the well-foundedness of \( \alpha_1 \) (or any other member).

  Therefore, \( \bigcup A \) is an ordinal.

  \SubProofOf{thm:def:ordinal/intersection} Let \( A \) be a nonempty collection of ordinals. \Cref{thm:def:hereditarily_transitive_set/intersection} implies that its intersection \( \bigcap A \) is hereditarily transitive.

  We must show that it is well-founded. Any nonempty subset \( B \) of the intersection is a subset of every ordinal in \( A \), so it has a minimal element.

  Therefore, \( \bigcap A \) is an ordinal.

  \SubProofOf{thm:def:ordinal/transitive_set}

  \SufficiencySubProof* By \cref{thm:def:ordinal/recursion}, the members of every ordinal are ordinals.

  \NecessitySubProof* Suppose that \( A \) is a transitive set of ordinals. We must show that it is itself an ordinal.

  Since \( A \) is hereditarily transitive, it is sufficient to prove that it is well-founded. This is straightforward enough --- if \( A \) contains an infinitely descending chain
  \begin{equation*}
    \cdots \in \alpha_n \in \cdots \in \alpha_2 \in \alpha_1,
  \end{equation*}
  this contradicts the well-foundedness of \( \alpha_1 \) (or any other member).

  \SubProofOf{thm:def:ordinal/closure} Let \( A \) be a collection of ordinals. \Cref{thm:closure_of_collection_of_hereditarily_transitive_sets} implies that the closure \( \cl^T(A) \) is a (hereditarily) transitive set.

  We will show that its members are ordinals, which, given \cref{thm:def:ordinal/transitive_set}, would imply that \( \cl^T(A) \) is an ordinal.

  Fix any member \( x \) of \( \cl^T(A) \). There exists some \( n \) such that \( x \) belongs to \( {\bigcup}^n A \). But \( {\bigcup}^n A \) is a set of ordinals, as can be shown via \fullref{thm:omega_recursion} on \( n \):
  \begin{itemize}
    \item This is vacuous for \( A = {\bigcup}^0 A \).
    \item Suppose that \( {\bigcup}^n A \) is a set of ordinals. Then \cref{thm:def:ordinal/union} implies that \( {\bigcup}^{n+1} A \) is an ordinal, and \cref{thm:def:ordinal/recursion} implies that \( {\bigcup}^{n+1} A \) is a set of ordinals.
  \end{itemize}

  Generalizing on \( x \), we conclude that all members of \( \cl^T(A) \) are ordinals, and so \( \cl^T(A) \) is itself an ordinal.
\end{proof}

\begin{proposition}\label{thm:members_of_omega_are_ordinals}
  All members of the \hyperref[def:smallest_inductive_set]{smallest inductive set} \( \omega \) are \hyperref[def:ordinal]{ordinals}.
\end{proposition}
\begin{comments}
  \item Due to \fullref{thm:omega_is_model_of_pa}, we regard \( \omega \) as the set of \hyperref[def:natural_numbers]{natural numbers}. So this proposition states that the natural numbers are ordinals.
\end{comments}
\begin{proof}
  We will use \cref{thm:omega_induction}.
  \begin{itemize}
    \item The empty set \( \varnothing \) is vacuously transitive and well-ordered, so it is an ordinal.
    \item If \( n \) is an ordinal, \cref{thm:def:hereditarily_transitive_set/successor} implies that so is its successor \( S(n) \)
  \end{itemize}
\end{proof}

\begin{corollary}\label{thm:omega_is_an_ordinal}
  The \hyperref[def:smallest_inductive_set]{smallest inductive set} \( \omega \) is an \hyperref[def:ordinal]{ordinal}.
\end{corollary}
\begin{proof}
  \Cref{thm:omega_is_hereditarily_transitive} implies that \( \omega \) is transitive, and \cref{thm:members_of_omega_are_ordinals} implies that its members are ordinals. Then, by \cref{thm:def:ordinal/transitive_set}, \( \omega \) is itself an ordinal.
\end{proof}

\begin{theorem}[Burali-Forti paradox]\label{thm:burali_forti_paradox}
  There is no set of all \hyperref[def:ordinal]{ordinals}.
\end{theorem}
\begin{comments}
  \item This paradox arises from self-reference; see \cref{ex:self_reference/russells_paradox}.

  \item We can still speak about the \hyperref[con:set_theory_class]{class} of all ordinals --- see \cref{def:class_of_all_ordinals}.
\end{comments}
\begin{proof}
  Suppose that \( \synop*{On} \) is the set of all ordinals. Then \( \synop*{On} \) is transitive because, by \cref{thm:def:hereditarily_transitive_set/recursion}, the members of each ordinal in \( \synop*{On} \) are themselves ordinals, and thus also belong to \( \synop*{On} \). \Cref{thm:def:ordinal/transitive_set} implies that \( \synop*{On} \) is an ordinal, and thus belongs to itself.

  But \cref{thm:def:hereditarily_transitive_set/successor} implies that the successor \( S(\synop*{On}) \) is also an ordinal, and thus membership in \( S(\synop*{On}) \) is irreflexive. But this leads to a contradiction because \( \synop*{On} \) belongs to itself.
\end{proof}

\begin{definition}\label{def:class_of_all_ordinals}\mcite[notation I.7.7]{Kunen2013SetTheory}
  \Fullref{thm:burali_forti_paradox} shows that, assuming the \hyperref[def:zfc/foundation]{axiom of foundation}, the existence of a set containing all \hyperref[def:ordinal]{ordinals} leads to a contradiction. Still, we may find it convenient to speak about the \hyperref[con:set_theory_class]{class} \( \synop*{On} \) of all ordinals. We can formally introduce it as
  \begin{equation*}
    \synop{On} \coloneqq \set{ \synx \given \synpred{ordinal}(\synx) }.
  \end{equation*}

  We have used the \( \synpred{ordinal} \) predicate from \cref{def:ordinal}.
\end{definition}

\paragraph{Ordinal ordering}

\begin{proposition}\label{thm:ordinal_subset_inclusion}
  For \hyperref[def:ordinal]{ordinals}, set membership coincides with strict \hyperref[def:inclusion_order]{subset inclusion}:
  \begin{equation*}
    \alpha \in \beta \T{if and only if} \alpha \subsetneq \beta.
  \end{equation*}
\end{proposition}
\begin{proof}
  \SufficiencySubProof If \( \alpha \in \beta \), transitivity of \( \beta \) implies that \( \alpha \subseteq \beta \), while \cref{thm:simple_foundation_theorems/member_of_itself} implies that \( \alpha \neq \beta \).

  \NecessitySubProof\mcite[lemma I.7.10]{Kunen2013SetTheory} Suppose that \( \alpha \subsetneq \beta \). Let \( \gamma \) be the smallest member of \( \beta \setminus \alpha \) with respect to membership. We will prove that \( \alpha = \gamma \).

  To show that \( \gamma \) is a subset of \( \alpha \), suppose that \( \gamma \setminus \alpha \) is nonempty, and fix its smallest member \( \delta \). Then \( \alpha \) and \( \delta \) are comparable, and we have three possibilities:
  \begin{itemize}
    \item If \( \alpha \in \delta \), then \( \alpha \in \gamma \) since \( \gamma \) is transitive. By \cref{thm:simple_foundation_theorems/member_of_itself}, \( \alpha \not\in \alpha \), so \( \alpha \in \gamma \setminus \alpha \), which contradicts the minimality of \( \delta \).

    \item If \( \alpha = \delta \), similarly \( \alpha \in \gamma \), and we reach the same contradiction.

    \item If \( \delta \in \alpha \), this contradicts the choice of \( \delta \) as a member of \( \gamma \setminus \alpha \).
  \end{itemize}

  We conclude that \( \gamma \setminus \alpha \) is empty, and, by \cref{thm:set_difference/subset}, \( \gamma \) is a subset of \( \alpha \). Conversely, fix some \( \delta \) in \( \alpha \). As a member of \( \beta \), is comparable with \( \gamma \), and we again have several possibilities:
  \begin{itemize}
    \item If \( \gamma \in \delta \), then \( \gamma \in \alpha \) since \( \alpha \) is transitive. But this contradicts our choice of \( \gamma \) as a member of \( \beta \setminus \alpha \).

    \item If \( \gamma = \delta \), obviously \( \gamma \in \alpha \), and we again obtain a contradiction.

    \item It remains for \( \delta \in \gamma \) to hold.
  \end{itemize}

  Therefore \( \alpha = \gamma \), and thus \( \alpha \in \beta \).
\end{proof}

\begin{proposition}\label{thm:ordinal_trichotomy}\mcite[thm. I.7.11(3)]{Kunen2013SetTheory}
  Any two \hyperref[def:ordinal]{ordinals} are \hyperref[def:partially_ordered_set/comparable]{comparable} via set membership.
\end{proposition}
\begin{proof}
  Fix ordinals \( \alpha \) and \( \beta \). \Cref{thm:def:ordinal/intersection} implies that their intersection \( \alpha \cap \beta \) is also an ordinal. We have several possibilities:
  \begin{itemize}
    \item If \( \alpha \cap \beta = \alpha = \beta \), then \( \alpha = \beta \).
    \item If \( \alpha \cap \beta = \alpha \subsetneq \beta \), \cref{thm:ordinal_subset_inclusion} implies that \( \alpha \in \beta \).
    \item If \( \alpha \cap \beta = \beta \subsetneq \alpha \), \cref{thm:ordinal_subset_inclusion} implies that \( \beta \in \alpha \).
    \item Finally, if \( \alpha \cap \beta \) is a strict subset of both \( \alpha \) and \( \beta \), \cref{thm:ordinal_subset_inclusion} implies that \( \alpha \cap \beta \) belongs to itself, which contradicts irreflexivity.
  \end{itemize}

  Therefore, \( \alpha \) and \( \beta \) are comparable.
\end{proof}

\begin{proposition}\label{thm:ordinal_ordering}
  The \hyperref[def:class_of_all_ordinals]{class of all ordinals} is itself \hyperref[def:well_ordered_set]{well-ordered} by set membership.

  More precisely, in \hyperref[def:zfc]{\logic{ZFC}}, the axioms for well-ordered relations can be derived for membership with all quantifiers \hyperref[con:predicate_logic/relativization]{relativized} to ordinals.
\end{proposition}
\begin{proof}
  We will show that \( \in \), relativized to ordinals, is strict.

  \SubProofOf[def:binary_relation/irreflexive]{irreflexivity} If \( \alpha \) is an ordinal \Cref{thm:def:hereditarily_transitive_set/successor} implies that its successor \( S(\alpha) = \alpha \cup \set{ \alpha } \) is an ordinal, so membership in \( S(\alpha) \) is irreflexive and \( \alpha \) cannot belong to itself.

  \SubProofOf[def:binary_relation/transitive]{transitivity} If \( \alpha \in \beta \in \gamma \) for some ordinals, then \( \alpha \in \gamma \) since \( \gamma \) is by definition a transitive set.

  \SubProofOf[def:totally_ordered_set/trichotomic]{trichotomy} Follows from \cref{thm:ordinal_trichotomy}.

  \SubProofOf[def:well_founded_relation]{well-foundedness} Any descending chain
  \begin{equation*}
    \cdots \in \alpha_n \in \cdots \in \alpha_2 \in \alpha_1
  \end{equation*}
  of ordinals leads to a contradiction with the well-foundedness of \( \alpha_1 \) (or any other member).
\end{proof}

\begin{definition}\label{def:ordinal_ordering}\mcite[19]{Jech2006SetTheory}
  Specifically for \hyperref[def:ordinal]{ordinals}, we will denote set membership by the traditional \hyperref[eq:def:peano_arithmetic_ordering/strict]{strict order} symbol \( {<} \), and also use the corresponding nonstrict order \( {\leq} \). We can regard it as a relation on the \hyperref[def:class_of_all_ordinals]{class of all ordinals}.
\end{definition}
\begin{comments}
  \item According to \cref{thm:ordinal_ordering}, when restricted to ordinals, the nonstrict relation \( {\leq} \) is a well-order, and according to \cref{thm:ordinal_subset_inclusion}, it coincides with \hyperref[def:inclusion_order]{set inclusion}.

  \item As we will show in \cref{thm:def:ordinal_ordering/extrema}, suprema and infima of ordinals are given by unions and intersections.
\end{comments}

\begin{proposition}\label{thm:def:ordinal_ordering}
  The \hyperref[def:ordinal_ordering]{ordinal ordering} has the following basic properties:
  \begin{thmenum}
    \thmitem{thm:def:ordinal_ordering/global_initial_segment} Every ordinal equals the set of all strictly smaller ordinals\fnote{Formally, this is obvious since ordinals are ordered by set membership. However, the statement is still sometimes useful.}.

    \thmitem{thm:def:ordinal_ordering/bounded_initial_segment} For every ordinal \( \alpha \), every member \( \beta \) equals its \hyperref[def:order_interval/unbounded]{open initial segment} \( \alpha_{<\beta} \).

    \thmitem{thm:def:ordinal_ordering/extrema} Every nonempty set of ordinals has \hyperref[def:extremal_points/supremum_and_infimum]{suprema and minima} given by unions and intersections.

    \thmitem{thm:def:ordinal_ordering/successor} The \hyperref[def:ordinal_successor]{successor} \( S(\alpha) = \alpha \cup \set{ \alpha } \) of an ordinal \( \alpha \) \hyperref[def:partial_order_covering]{covers} \( \alpha \), in the sense that \( \beta > \alpha \) implies \( \beta \geq S(\alpha) \).
  \end{thmenum}
\end{proposition}
\begin{proof}
  \SubProofOf{thm:def:ordinal_ordering/global_initial_segment} Trivial.

  \SubProofOf{thm:def:ordinal_ordering/bounded_initial_segment} Consider the initial segment
  \begin{equation*}
    \alpha_{<\beta} = \set{ \gamma \in \alpha \given \gamma \in \beta }.
  \end{equation*}

  Clearly \( \alpha_{<\beta} = \alpha \cap \beta \). Given that \( \alpha \) is a transitive set, however, we have \( \beta \subseteq \alpha \) and thus
  \begin{equation*}
    \alpha_{<\beta} = \alpha \cap \beta = \beta.
  \end{equation*}

  \SubProofOf{thm:def:ordinal_ordering/extrema} By \cref{thm:def:ordinal/union} and \cref{thm:def:ordinal/intersection}, unions and intersections of nonempty sets of ordinals are also ordinals. They act as \hyperref[def:extremal_points/supremum_and_infimum]{suprema and minima} in the \( \synop*{On} \), since, by \cref{thm:binary_union_and_intersection_as_operations}, the same operations are suprema and minima in \hyperref[def:lattice_of_sets]{lattices of sets}.

  \SubProofOf{thm:def:ordinal_ordering/successor} \Cref{thm:def:ordinal/successor} implies that \( S(\alpha) = \alpha \cup \set{ \alpha } \) is an ordinal.

  Suppose that \( \alpha < \beta \). By \cref{thm:ordinal_trichotomy}, \( S(\alpha) \) and \( \beta \) are comparable. Note that \( \beta < S(\alpha) \) leads to \( \beta \leq \alpha \), which contradicts the well-foundedness of \( \alpha \), so this leaves us with \( S(\alpha) \leq \beta \).
\end{proof}

\begin{remark}\label{rem:def:ordinal_successor}
  It follows from \cref{thm:def:ordinal_ordering/global_initial_segment} and \cref{thm:def:ordinal_ordering/successor} that, for any ordinal \( \alpha \), we have
  \begin{align*}
    \alpha    &= \set{ \beta \given \beta \T{is an ordinal and} \beta < \alpha }, \\
    S(\alpha) &= \set{ \beta \given \beta \T{is an ordinal and} \beta \leq \alpha }.
  \end{align*}

  This shows that the \hyperref[def:ordinal_successor]{ordinal successor} operation arises naturally in the context of ordinals.
\end{remark}

\begin{definition}\label{def:successor_and_limit_ordinal}\mcite[20]{Jech2006SetTheory}
  We say that the ordinal \( \alpha \) is a \term[ru=непредельный ординал (\cite[84]{ШеньВерещагин2020НачалаТеорииМножеств})]{successor ordinal} if there exists an ordinal \( \beta \) such that \( \alpha = S(\beta) \). Otherwise, we say that \( \alpha \) is a \term[ru=предельный ординал (\cite[84]{ШеньВерещагин2020НачалаТеорииМножеств})]{limit ordinal}.
\end{definition}
\begin{comments}
  \item Note that \( 0 \) is considered a limit ordinal in some books like
  \cite[20]{Jech2006SetTheory} and
  \cite[84]{ШеньВерещагин2020НачалаТеорииМножеств}
  but not in other like
  \cite[def. I.7.15]{Kunen2013SetTheory},
  \cite[203]{Enderton1977SetTheory} and
  \cite[32]{КусраевКутателадзе2005БулевозначныйАнализ}. For disambiguation, we use the phrase \enquote{nonzero limit ordinal}.

  \item See \cref{thm:limit_ordinal_order_topology} for a topological and \cref{thm:limit_ordinal_division_characterization} for an arithmetic characterization of limit ordinals.

  \item These notions should not be confused with \hyperref[def:successor_and_limit_cardinal]{successor and weak/strong limit cardinals}.
\end{comments}

\paragraph{Transfinite induction}

\begin{theorem}[Transfinite induction]\label{thm:transfinite_induction}
  As shown in \cref{thm:members_of_omega_are_ordinals}, \hyperref[def:ordinal]{ordinals} generalize \hyperref[def:natural_numbers]{natural numbers}. Correspondingly, \fullref{thm:strong_induction_on_natural_numbers} generalizes to ordinals:
  \begin{equation}\label{eq:thm:transfinite_induction/sol}
    \overbrace{\qforall {\synp^{\syn\iota \synimplies \syn\omicron}}}^{\synp \T*{is a predicate}}
      (
        \qforall {\syna^{\syn\iota}}
          (
            \synpred{ordinal}(\syna)
            \synimplies
            \overbrace
              {
                (
                  \underbrace{ \qforall {\synb \synin \syna} \synp \synb }_{\mathclap{\substack{\T{inductive} \\ \T{hypothesis}}}}
                  \synimplies
                  \underbrace{ \synp \syna }_{\mathclap{\substack{\T{inductive step} \\ \T{conclusion}}}}
                )
              }^{\T{inductive step}}
          )
        \synimplies
        \underbrace
          {
            \qforall {\syna^{\syn\iota}} (\synpred{ordinal}(\syna) \synimplies \synp \syna)
          }_{\T{conclusion}}
      ).
  \end{equation}

  Informally, the principle can be restated as follows:
  \begin{displayquote}
    In order for \( p(\alpha) \) to hold \emph{for every} ordinal \( \alpha \), it is sufficient for \( p(\alpha) \) to hold \emph{whenever} \( p(\beta) \) holds for every \( \beta \in \alpha \).
  \end{displayquote}

  Its \hyperref[def:fol_adaptation_of_monadic_sol_theory]{first-order adaptation} is
  \begin{equation}\label{eq:thm:transfinite_induction}
    \qforall \syna
      (
        \synpred{ordinal}(\syna)
        \synimplies
        \overbrace
          {
            (
              \underbrace{ \qforall {\synb \synin \syna} \syn\varphi[\synx \synsubst \synb] }_{\mathclap{\substack{\T{inductive} \\ \T{hypothesis}}}}
              \synimplies
              \underbrace{ \syn\varphi[\synx \synsubst \syna] }_{\mathclap{\substack{\T{inductive step} \\ \T{conclusion}}}}
            )
          }^{\T{inductive step}}
      )
    \synimplies
    \underbrace
      {
        \qforall \syna (\synpred{ordinal}(\syna) \synimplies \syn\varphi[\synx \synsubst \syna])
      }_{\T{conclusion}}.
  \end{equation}

  This is a formula schema in the sense of \cref{def:fol_formula_schema}. We require the \hyperref[con:schemas_and_instances]{instances} of \( \syna \) and \( \synb \) not to be free in \( \syn\varphi \), but otherwise put no restrictions on the placeholders. Note that we must take the \hyperref[def:fol_quantifier_closure]{universal closure} of each concrete axiom of the schema.
\end{theorem}
\begin{comments}
  \item \hyperref[def:successor_and_limit_ordinal]{Successor and limit ordinals} enable a useful reformulation --- see \cref{rem:transfinite_induction_with_limit_ordinals}.
\end{comments}
\begin{proof}
  Special case of \fullref{thm:epsilon_induction}, with mundane simplifications, applied to the predicate
  \begin{equation*}
    \qabs {\syna^{\syn\iota}} (\synpred{ordinal}(\syna) \synimplies \synp \syna).
  \end{equation*}
\end{proof}

\begin{corollary}[Bounded transfinite induction]\label{thm:bounded_transfinite_induction}
  We can restrict \fullref{thm:transfinite_induction} to an \hyperref[def:order_interval/unbounded]{initial segment} of ordinals:
  \begin{equation}\label{eq:thm:bounded_transfinite_induction/sol}
    \qforall {\synp^{\syn\iota \synimplies \syn\omicron}} \qforall {\syna^{\syn\iota}}
      \parens[\big]
      {
        \synpred{ordinal}(\syna)
        \synimplies
        (
          \qforall {\synb \synin \syna}
            (
              \overbrace
                {
                  (
                    \underbrace{ \qforall {\sync \synin \synb} \synp \sync }_{\mathclap{\substack{\T{inductive} \\ \T{hypothesis}}}}
                    \synimplies
                    \underbrace{ \synp \synb }_{\mathclap{\substack{\T{inductive step} \\ \T{conclusion}}}}
                  )
                }^{\T{inductive step}}
            )
          \synimplies
          \underbrace
            {
              \qforall {\synb \synin \syna} \synp \synb
            }_{\T{conclusion}}
        )
      }.
  \end{equation}

  For every formula \( \varphi \) in the language of set theory not containing \( \tau \), \( \syny \) nor \( \synz \) as free variables, the following is a theorem of \logic{ZFC}:
  \begin{equation}\label{eq:thm:bounded_transfinite_induction}
    \qforall \syna
      \parens[\big]
      {
        \synpred{ordinal}(\syna)
        \synimplies
        (
          \qforall {\synb \synin \syna}
            (
              \overbrace
                {
                  (
                    \underbrace{ \qforall {\sync \synin \synb} \syn\varphi[\synx \synsubst \sync] }_{\mathclap{\substack{\T{inductive} \\ \T{hypothesis}}}}
                    \synimplies
                    \underbrace{ \syn\varphi[\synx \synsubst \synb] }_{\mathclap{\substack{\T{inductive step} \\ \T{conclusion}}}}
                  )
                }^{\T{inductive step}}
            )
          \synimplies
          \underbrace
            {
              \qforall {\synb \synin \syna} \syn\varphi[\synx \synsubst \synb]
            }_{\T{conclusion}}
        )
      }.
  \end{equation}

  See the comments in \cref{def:peano_arithmetic/theory/PA3} regarding variables and quantification in axiom schemas and \cref{con:induction} for a general discussion of induction.

  See \cref{rem:transfinite_induction_with_limit_ordinals} about a reformulation that is often useful in practice.
\end{corollary}
\begin{proof}
  Special case of \fullref{thm:transfinite_induction} applied to the predicate
  \begin{equation*}
    \qabs {\synb^{\syn\iota}} (\synb \synin \syna \synimplies \synp \synb).
  \end{equation*}
\end{proof}

\begin{definition}\label{def:transfinite_sequence}\mcite[21]{Jech2006SetTheory}
  We can generalize \hyperref[def:sequence]{natural number sequences} to \hyperref[def:ordinal]{ordinals} by considering \hyperref[def:indexed_family]{indexed families} whose domains are arbitrary ordinals rather than the \hyperref[def:smallest_inductive_set]{smallest inductive set} \( \omega \). We call them \term{transfinite sequences}.
\end{definition}

\begin{definition}\label{def:truncated_sequence}\mimprovised
  Given an \hyperref[def:ordinal]{ordinal} \( \alpha \), for every \( \beta < \alpha \), we say that the \hyperref[def:transfinite_sequence]{transfinite sequence} \( \seq{ x_\gamma }_{\gamma < \beta} \) is a \term{truncation} of \( \seq{ x_\gamma }_{\gamma < \alpha} \).
\end{definition}

\begin{theorem}[Transfinite recursion]\label{thm:transfinite_recursion}\mcite[thm. 2.15]{Jech2006SetTheory}
  Fix an \hyperref[def:ordinal]{ordinal} \( \alpha \) and a nonempty set \( A \). Suppose that we are given some transformation \( T: \pow(\alpha \times A) \to A \), which sends the graph of a \hyperref[def:set_valued_function]{set-valued function} from \( \alpha \) to \( A \) into \( A \). Then there exists a unique \( \alpha \)-indexed \hyperref[def:transfinite_sequence]{transfinite sequence} \( f: \alpha \to A \) such that, for every \( \beta \) in \( \alpha \), we have \( f(\beta) = T(\gr f\restr_\beta) \).
\end{theorem}
\begin{comments}
  \item This is a vast generalization of \fullref{thm:omega_recursion} from \hyperref[def:sequence]{natural number sequences} to \hyperref[def:transfinite_sequence]{transfinite sequences}.

  \item \hyperref[def:successor_and_limit_ordinal]{Successor and limit ordinals} enable a useful reformulation --- see \cref{rem:transfinite_induction_with_limit_ordinals}.
\end{comments}
\begin{proof}
  The proof is analogous to that of \fullref{thm:omega_recursion}, but we will give it anyway to highlight the difference between using \fullref{thm:omega_induction} and \fullref{thm:bounded_transfinite_induction}.

  Let \( G \subseteq \pow(\alpha \times A) \) be the set of all \hyperref[def:partial_function]{partial single-valued functions} \( g: \alpha \to A \) such that
  \begin{itemize}
    \item There exists some \( \beta_g \in \alpha \) such that \( g \) is defined only in the \hyperref[def:order_interval/unbounded]{open initial segment} \( \alpha_{< \beta_g} \). That is, \( g \) is defined for all \( \beta \) up to not including \( \beta_g \).

    \item \( g(\beta) = T(\gr g\restr_\beta) \) for all \( \beta < \beta_g \).
  \end{itemize}

  Clearly \( G \) is nonempty because the function \( \set{ (\varnothing, T(\varnothing)) } \) belongs to \( G \).

  The conditions imposed on the functions in \( G \) ensure that each of them is defined in some \hyperref[def:order_interval/unbounded]{initial segment} of \( \alpha \).

  Define \( f \coloneqq \bigcup G \). At this point \( f \) is simply some \hyperref[def:function]{set-valued function}. We must now show that \( f \) has all the properties that we want.

  \SubProof{Proof that \( \dom f = \alpha \)} First, we will use \fullref{thm:bounded_transfinite_induction} to show that the \hyperref[def:set_valued_function/domain]{domain} of \( f \) is the entirety of \( \omega \).

  Fix \( \beta \in \dom f \). Then there exists a function \( g \in G \) defined for all \( \gamma < \beta \).

  \begin{itemize}
    \item If \( g \) is also defined at \( \beta \), this directly proves that \( \beta \) belongs to \( \dom f \).

    \item If \( g \) is not defined at \( \beta \), consider
    \begin{equation*}
      \widehat g \coloneqq g \cup \set{ (\beta, T(\gr g\restr_\beta) }.
    \end{equation*}

    The function \( \widehat g \) is again a single-valued partial function and thus it belongs to \( G \), hence \( \beta \) belongs to \( \dom f \).
  \end{itemize}

  This completes the induction.

  \SubProofOf[def:function]{single-valuedness} Now that we know that \( f(\beta) \) is nonempty for any \( \beta < \alpha \), we must show that \( f(\beta) \) is a singleton set.

  We will again use induction. Fix any \( \beta < \alpha \) and suppose that \( f(\gamma) \) is a singleton set for every \( \gamma < \beta \). At least one partial function in \( G \) is defined at \( \beta \). Let \( g \) and \( h \) both be such (single-valued partial) functions.

  Then
  \begin{equation*}
    g(\beta) = T(\gr g\restr_\beta) = T(\gr f\restr_\beta) = T(\gr h\restr_\beta) = h(\beta),
  \end{equation*}
  hence \( g \) and \( h \) coincide at \( \beta \), which in turn implies that \( f \) is single-valued at \( \beta \).

  Therefore, transfinite induction allows us to conclude that \( f \) is a single-valued total function.

  \SubProofOf[def:function]{uniqueness} It remains to show that \( f \) is unique.

  We will again use induction on \( \beta < \alpha \) to show that \( f_1(\beta) = f_2(\beta) \). Indeed, suppose that for every \( \gamma < \beta \), we have \( f_1(\gamma) = f_2(\gamma) \). Then
  \begin{equation*}
    f_2(\beta) = T(\gr f_1\restr_\beta)) = T(\gr f_2\restr_\beta) = f_2(\beta).
  \end{equation*}

  Therefore, \( \gr f_1 \) and \( \gr f_2 \) coincide as sets.
\end{proof}

\begin{remark}\label{rem:transfinite_induction_with_limit_ordinals}
  \hyperref[def:successor_and_limit_ordinal]{Successor and limit ordinals} enable a commonly used pattern for \fullref{thm:transfinite_induction} and related principles like \fullref{thm:bounded_transfinite_induction} and \fullref{thm:transfinite_recursion}.

  Informally, the principle of transfinite induction can be stated as follows:
  \begin{displayquote}
    In order for \( p(\alpha) \) to hold \emph{for every} ordinal \( \alpha \), it is sufficient for \( p(\alpha) \) to hold \emph{whenever} \( p(\beta) \) holds for every \( \beta \in \alpha \).
  \end{displayquote}

  Instead, the principle can be restated as follows:
  \begin{displayquote}
    In order for \( p(\alpha) \) to hold \emph{for every} ordinal \( \alpha \), it suffices for \( p(\alpha) \) to hold in the following three cases:
    \begin{itemize}
      \item In the base case, where \( \alpha = \varnothing \) is zero, we must simply prove \( p(\varnothing) \).

      \item If \( \alpha \) is a successor ordinal, then \( \alpha = S(\beta) \) and the inductive step requires proving that \( p(\alpha) \) holds assuming \( p(\beta) \) does.

      \item If \( \alpha \) is a limit ordinal, then \( \alpha = S(\beta) \) and the inductive step requires proving that \( p(\alpha) \) holds assuming \( p(\beta) \) does for every \( \beta < \alpha \).
    \end{itemize}
  \end{displayquote}

  The proofs are often distinct in these three cases, so this splitting becomes useful. See \cref{thm:ordinal_bijection_and_choice} and \cref{def:cumulative_hierarchy} for concrete examples.
\end{remark}
\begin{comments}
  \item This is akin to the relation between \fullref{thm:induction_on_natural_numbers} and \fullref{thm:strong_induction_on_natural_numbers}. Of course, as explained in \cref{thm:induction_on_natural_numbers}, we may need multiple base cases.
\end{comments}

\begin{concept}\label{con:induction}
  \term{Mathematical induction} is a very valuable proof technique for universal statements. The proof of \cref{thm:peano_arithmetic_predecessor_existence} contains remarks regarding its usage and its difference from deduction principles that are formalized via \hyperref[def:deduction_system]{deduction systems}. Although it is a logical tool, this remark belongs to this section because it contains several induction and recursion principles.

  More generally, given a \hyperref[def:fol_formula]{first-order formula} \( \varphi \) over some \hyperref[def:first_order_language]{first-order language}, certain \hyperref[def:fol_theory]{logical theories} allow us to prove indirectly \( \qforall \synx \varphi[\synx \mapsto \syny] \) by proving simpler statements (\cref{def:peano_arithmetic/theory/PA3} contains very important remarks regarding the free variables of \( \varphi \)). This can be done in cases where every \hyperref[def:fol_semantics/model]{model} \( X = (X, I) \) of the theory allows us to exhaust its universe \( X \) in a small finite number of steps. We can sometimes use the same steps to instead build objects. The latter principles is called \term{recursion}.

  It should be noted that induction and recursion are used interchangeably in the literature.

  Not much more can be said at this level of generality, so we list several induction principles and give examples of their usage:
  \begin{thmenum}
    \thmitem{con:induction/peano_arithmetic} The most basic induction principles is the (weak) natural number induction. It is best described via the axiom schema \eqref{eq:def:peano_arithmetic/theory/PA3}. \Cref{thm:peano_arithmetic_predecessor_existence} contains detailed commentary regarding its usage and most of the proofs in \fullref{sec:natural_numbers} are performed inductively.

    It its set-theoretic form \fullref{thm:omega_induction} it is important as a tool for introducing a model of Peano arithmetic. It is used directly for proving \cref{thm:omega_is_hereditarily_transitive} and \fullref{thm:omega_recursion}. It is essentially the same as \cref{rem:transfinite_induction_with_limit_ordinals} without \cref{rem:transfinite_induction_with_limit_ordinals/limit_case}.

    \Fullref{thm:omega_recursion} is an important standalone tool that allows us to perform recursive definitions for natural numbers. The latter is used, often implicitly, in a great variety of places, from the ability to define natural number operations in \cref{def:omega_operations} to the definition of semigroup exponentiation in \cref{def:semigroup/exponentiation}.

    \thmitem{con:induction/well_founded} A vast generalization of natural number induction is \fullref{thm:well_founded_induction}. It is stated in a very general setting, but is not frequently used. It can be used to prove \fullref{thm:epsilon_induction}, which however is even less frequently used. We do not use neither in practice, however the special case where \( X = \BbbN \) is called strong induction on natural numbers.

    The usual (weak) natural number induction which is performed by proving the statement for \( 0 \) and then proceeding to prove it for \( n + 1 \) by assuming that it holds for \( n \). Strong induction instead has no base cases and is performed by proving a statement for \( n \) by assuming that it holds for all natural numbers strictly smaller than \( n \).

    Well-founded induction and \( \varepsilon \)-induction have no corresponding recursion principle.

    \thmitem{con:induction/transfinite} Another vast generalization of natural number induction is \fullref{thm:bounded_transfinite_induction}, which is further generalized by \fullref{thm:transfinite_induction}.

    Both principles are used to prove fundamental properties of the ordinals. Outside of set theory, transfinite induction is immensely useful, but it is rarely used directly. Instead, it is usually combined with the \hyperref[def:zfc/choice]{axiom of choice} via \fullref{thm:zorns_lemma}. \Cref{rem:transfinite_induction_with_limit_ordinals} contains notes on how it is used directly.

    Only bounded transfinite induction has a corresponding recursion principle --- \fullref{thm:transfinite_recursion}. Unbounded transfinite induction cannot define such a principle because that would easily lead to \fullref{thm:burali_forti_paradox}. \Cref{rem:unbounded_transfinite_recursion} shows how to circumvent this, however.

    Transfinite recursion is used to construct the cumulative hierarchy in \cref{def:cumulative_hierarchy}.

    \Cref{rem:cardinal_transfinite_recursion_and_induction} provides alternative transfinite recursion and induction principles for cardinals rather than for ordinals.

    \thmitem{con:induction/grammars} Within this monograph, we use \fullref{thm:induction_on_abstract_syntax}, for example \hyperref[def:propositional_formula]{propositional} and \hyperref[def:fol_formula]{first-order formulas}. It is not an established principle like the others.

    Rather than having a corresponding recursion principle, we construct words via \hyperref[def:formal_grammar]{formal grammars}.

    \thmitem{con:induction/lattices} \Fullref{thm:knaster_tarski_theorem} and \fullref{thm:knaster_tarski_iteration} hold in \hyperref[def:complete_lattice]{complete lattices}, and enable more esoteric principles like \fullref{thm:induction_on_recursively_defined_relations}.
  \end{thmenum}
\end{concept}

\paragraph{Order types}

\begin{proposition}\label{thm:ordinal_isomorphism}
  Two ordinals are equal if and only if they are order-isomorphic.
\end{proposition}
\begin{proof}
  \SufficiencySubProof Trivial.

  \NecessitySubProof Let \( \alpha \) and \( \beta \) be two ordinals. The case \( \alpha = \beta \) is clear. Without loss of generality, suppose that \( \beta < \alpha \). \Cref{thm:ordinal_subset_inclusion} implies that \( \beta \subsetneq \alpha \).

  Let \( f: \alpha \to \beta \) be an order isomorphism. Let \( \gamma_0 \) be the smallest value in \( \alpha \setminus \beta \).

  From \cref{thm:def:well_ordered_set/embedding_extensional} it follows that \( \gamma_0 \leq f(\gamma_0) \). But \( f(\gamma_0) \in \beta \), hence \( \gamma_0 < \beta \). But this contradicts our choice of \( \gamma_0 \) as a member of \( \alpha \setminus \beta \).

  Therefore, \( \gamma_0 \in \beta \). Since \( \gamma_0 \) was arbitrary, we conclude that \( \beta < \alpha \) leads to a contradiction.

  We can obtain an analogous contradiction for \( \alpha < \beta \), hence it only remains possible for \( \alpha \) and \( \beta \) to be equal.
\end{proof}

\begin{lemma}[Hartogs' lemma]\label{thm:hartogs_lemma}
  For every set \( A \) there exists a smallest ordinal \( \alpha \) such that no function from \( \alpha \) to \( A \) is injective.
\end{lemma}
\begin{proof}
  Define the set
  \begin{equation*}
    W \coloneqq \set{ (P, \leq) \given P \subseteq A \T*{and} \leq \T*{well-orders} P }.
  \end{equation*}

  Let \( \beta \) be an ordinal for which there exists an injective function \( f: \beta \to A \). At least one such pair of a ordinal and function exists because \( \varnothing: 0 \to A \) is an injective function.

  The image of \( f \) can be well-ordered by
  \begin{equation*}
    f(\delta) < f(\gamma) \T{if and only if} \delta \in \gamma,
  \end{equation*}
  where \( \delta \) and \( \gamma \) are members of \( \beta \). Thus, if we restrict the codomain of \( f \) to its image, it would be an explicit order isomorphism of \( (\beta, \in) \) and \( (\img f, <) \).

  We have shown that every ordinal \( \beta \) and every injective function \( f: \beta \to A \) induces a well-ordered set which belongs to \( W \). Furthermore, if \( f_1: \beta_1 \to A \) and \( f_2: \beta_2 \to A \) are two such injective functions and if the induced well-ordered sets \( (\img f_1, \leq_1) \) and \( (\img f_2, \leq_2) \) are order-isomorphic, then \( \beta_1 \) and \( \beta_2 \) are themselves order-isomorphic and thus \( \beta_1 = \beta_2 \) by \cref{thm:ordinal_isomorphism}.

  Therefore, for any well-ordered set in \( W \) there exists at most one ordinal that induces it via some function. Let \( W' \) be the subset of all well-ordered sets in \( W \) induced by exactly one ordinal.

  We have that \( W' \) is a set and for each member of \( W' \) there corresponds exactly one ordinal. It follows from the \hyperref[def:zfc/replacement]{axiom schema of replacement} that these ordinals form a set. Denote this set by \( B \).

  There is no set of all ordinals by \fullref{thm:burali_forti_paradox}, hence there must exist an ordinal not in \( B \). But every ordinal that has an injective function into \( A \) belongs to \( B \). Hence, there exists some ordinal \( \alpha \) such that no function from \( \alpha \) to \( A \) is injective.

  If \( \alpha \) is not the smallest ordinal with this property, we can now easily take the smallest member of \( \alpha \) with this property.
\end{proof}

\begin{lemma}\label{thm:ordinal_bijection_and_choice}\mcite{MathSE:order_type_existence}
  Let \( A \) be a nonempty set and let \( c \) be a \hyperref[def:choice_function]{choice function} on \( A \). Then there exists an ordinal \( \alpha \) and a bijection between \( A \) and \( \alpha \).
\end{lemma}
\begin{proof}
  We will explicitly build the desired ordinal. We will use \fullref{thm:transfinite_recursion} in its structured form presented in \cref{rem:transfinite_induction_with_limit_ordinals} to build a \hyperref[def:transfinite_sequence]{transfinite sequence} of injective maps from ordinals into \( A \).

  Let \( \delta \) be the smallest ordinal such that no function from \( \delta \) to \( A \) is injective. Such an ordinal exists by \fullref{thm:hartogs_lemma}. Note that \( \delta \) cannot be zero because the empty function is always injective.

  Any smaller than \( \delta \) ordinal has at least one injective function into \( A \). We will use recursion bounded by \( \delta \) to construct a \hyperref[def:transfinite_sequence]{transfinite sequence} \( \seq{ \iota_\beta }_{\beta < \delta} \) satisfying the invariant that for any \( \beta \in \delta \), the function \( \iota_\beta \) is injective and strictly contains \( \iota_\gamma \) for any \( \gamma < \beta \).

  \begin{itemize}
    \item The zero ordinal has only one possible injective function into \( A \), namely the empty function \( \varnothing: 0 \to P \).

    \item Now let \( \iota_\beta: \beta \to A \) be an injective function.

    If \( \iota_\beta \) is surjective, then it is the desired bijection and the rest of the construction is irrelevant.

    If \( \iota_\beta \) is not surjective, define
    \begin{equation*}
      \begin{aligned}
        &\iota_{S(\beta)}: S(\beta) \to P \\
        &\iota_{S(\beta)}(\gamma) \coloneqq \begin{cases}
          \iota_\beta(\gamma),         &\gamma < \beta \\
          c(A \setminus \img(\iota_\beta)), &\gamma = \beta.
        \end{cases}
      \end{aligned}
    \end{equation*}

    This function is clearly injective.

    \item Let \( \lambda \) be a limit ordinal and let \( \iota_\beta \) be an injective function for any \( \beta < \lambda \). Note that the domain of \( \iota_\beta \) is not necessarily \( \beta \) --- it may be a smaller ordinal \( \gamma \) in case \( \iota_\gamma \) is surjective (and thus the desired bijection).

    In any case, we have a family \( \set{ \iota_\beta }_{\beta < \lambda} \) of functions such that \( \iota_\gamma \subseteq \iota_\beta \) if and only if \( \gamma < \beta \). We simply take their union
    \begin{equation*}
      \iota_\lambda \coloneqq \bigcup \set{ \iota_\beta \given \beta < \lambda }.
    \end{equation*}

    This function is injective because each of the constituent functions it injective.
  \end{itemize}

  We will now thoroughly build the order isomorphism itself.

  Let \( \alpha \leq \delta \) be the (nonstrict) initial segment of \( \delta \) such that \( \iota_\beta \) is fully defined on \( \beta \) for every \( \beta \in \alpha \). More precisely, let
  \begin{equation*}
    \alpha \coloneqq \set{ \beta \in \delta \given \dom(\iota_\beta) = \beta }.
  \end{equation*}

  We will show that \( \alpha \) is a successor ordinal. Note that \( \alpha \) cannot be zero because \( \dom(\iota_0) = \varnothing = 0 \). Aiming at a contradiction, assume that \( \alpha \) is a limit ordinal.

  Since for every \( \beta \in \alpha \) the value \( \iota_{S(\beta)}(\beta) \) is defined, we can conclude that the union
  \begin{equation*}
    \bigcup\set{ \iota_\beta \given \beta \in \alpha }
  \end{equation*}
  is an injective function from \( \alpha \) to \( A \).

  If \( \alpha = \delta \), this contradicts our choice of \( \delta \). If \( \alpha \in \delta \), this means that the function \( \iota_\alpha \) is equal to \( \iota_\gamma \) for some \( \gamma < \alpha \). But then \( \iota_{S(\gamma)} \) is not defined on the entirely of \( \gamma \). This implies that \( \gamma \geq \alpha \), which contradicts our assumption that \( \gamma < \alpha \).

  Therefore, \( \alpha \) cannot be a limit ordinal. It remains for \( \alpha \) to be a successor ordinal. Then there exists \( \beta \) such that \( \alpha = S(\beta) \).

  Suppose that \( \iota_\beta \) is not surjective. We can then construct \( \iota_{S(\beta)} \) as in the corresponding recursion step. This will be an injective function from \( \alpha = S(\beta) \) to \( A \). But then we would have \( \alpha \in \alpha \), which contradicts \cref{thm:simple_foundation_theorems/member_of_itself}.

  Therefore, \( \iota_\beta \) is a surjective function.
\end{proof}

\begin{theorem}\label{thm:well_ordered_order_type_existence}
  Any \hyperref[def:well_ordered_set]{well-ordered set} \( (P, \leq) \) is \hyperref[def:partially_ordered_set/morphism]{order-isomorphic} to a unique ordinal. This ordinal is called the \term{order type} of \( (P, \leq) \) and is denoted by \( \ord(P, \leq) \) or simply \( \ord(P) \).

  Furthermore, this isomorphism is unique.
\end{theorem}
\begin{proof}
  For \( P = \varnothing \), then the empty function \( \iota: 0 \to P \) is the desired isomorphism.

  We use \cref{thm:ordinal_bijection_and_choice} on \( P \) with the choice function
  \begin{equation*}
    \begin{aligned}
      &c: \pow(P) \setminus \set{ \varnothing } \to P \\
      &c(B) \coloneqq \min B
    \end{aligned}
  \end{equation*}
  to obtain an ordinal \( \alpha \) and a bijection \( \iota: \alpha \to P \).

  The function \( \iota \) is directly verified to be an order embedding in each of the recursion cases. Therefore, \( \iota \) is a surjective order embedding, that is, an order isomorphism.

  Uniqueness of \( \iota \) follows from \cref{thm:def:well_ordered_set/unique_isomorphism}.
\end{proof}

\begin{theorem}[Zermelo's well-ordering theorem]\label{thm:well_ordering_theorem}\mcite[thm. 6.4.18(i)]{Hinman2005Logic}
  Any \hyperref[def:set]{set} can be \hyperref[def:well_ordered_set]{well-ordered}.
\end{theorem}
\begin{comments}
  \item Within \hyperref[def:zfc]{\logic{ZF}}, this theorem is equivalent to the \hyperref[def:zfc/choice]{axiom of choice}. We list equivalent results in \cref{thm:axiom_of_choice_equivalences}.
\end{comments}
\begin{proof}
  \ImplicationSubProof[def:zfc/choice]{the axiom of choice}[thm:well_ordering_theorem]{well-ordering theorem} The empty set is trivially well-ordered.

  Let \( A \) be any nonempty set. By the axiom of choice, there exists a \hyperref[def:choice_function]{choice function} \( c \) for \( A \). We use \cref{thm:ordinal_bijection_and_choice} on \( A \) and \( c \) to obtain an ordinal \( \alpha \) and a bijection \( \iota: \alpha \to A \). Since \( \alpha \) itself is well-ordered, then the relation
  \begin{equation*}
    x <_A y \T{if and only if} \iota^{-1}(x) <_\alpha \iota^{-1}(y)
  \end{equation*}
  defines a well-order on \( A \).

  \ImplicationSubProof[thm:well_ordering_theorem]{well-ordering theorem}[def:zfc/choice]{axiom of choice} Let \( A \) be any set and suppose that \( < \) well-orders \( A \).

  Define the function
  \begin{equation*}
    \begin{aligned}
      &c: \pow(A) \setminus \set{ \varnothing } \to A \\
      &c(B) \coloneqq \min B.
    \end{aligned}
  \end{equation*}

  In the case where \( A \) is the empty set, \( c \) is the empty function.

  It is clear that \( c(B) \in B \) for every subset \( B \) of \( A \). Therefore, \( c \) is a choice function for \( A \).
\end{proof}
