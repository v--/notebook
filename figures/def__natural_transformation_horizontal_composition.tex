  \end{tikzcd}

    {[H \bincirc G](A)} \ar[d, "\beta_{G(A)}" swap] \ar[r, "{[H \bincirc F](f)}"] & {[H \bincirc G](B)} \ar[d, "\beta_{G(B)}"] \\

  \end{tikzcd}
