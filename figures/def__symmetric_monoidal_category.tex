  \begin{tikzpicture}[commutative diagrams/every diagram]
    \node (P1) at (canvas polar cs:angle=180+0*60,x radius=15ex,y radius=10ex) {\( (A \otimes B) \otimes C \)};
    \node (P2) at (canvas polar cs:angle=180+1*60,x radius=15ex,y radius=10ex) {\phantom{ABC}};
    \node (P3) at (canvas polar cs:angle=180+2*60,x radius=15ex,y radius=10ex) {\phantom{ABC}};

    \node (P4) at (canvas polar cs:angle=180+3*60,x radius=15ex,y radius=10ex) {\( B \otimes (C \otimes A) \)};
    \node (P5) at (canvas polar cs:angle=180+4*60,x radius=15ex,y radius=10ex) {\phantom{ABC}};
    \node (P6) at (canvas polar cs:angle=180+5*60,x radius=15ex,y radius=10ex) {\phantom{ABC}};

    \begin{scope}[node distance=-2.9em]
      \node[left=of P2]  {\( (B \otimes A) \otimes C \)};
      \node[right=of P3] {\( B \otimes (A \otimes C) \)};
      \node[right=of P5] {\( (B \otimes C) \otimes A \)};
      \node[left=of P6]  {\( (B \otimes C) \otimes A \)};
    \end{scope}

    \path[commutative diagrams/.cd, every arrow, every label]

    (P1) edge node[swap, pos=0.4] {\( \beta_{A, B} \otimes \id_C \)} (P2)
    (P2) edge node[swap]          {\( \alpha_{B, A, C} \)}           (P3)
    (P3) edge node[swap, pos=0.6] {\( \id_B \otimes \beta_{A, C} \)} (P4)

    (P1) edge node[pos=0.4]       {\( \alpha_{A, B, C} \)}           (P6)
    (P6) edge node                {\( \beta_{A, B \otimes C} \)}     (P5)
    (P5) edge node[pos=0.6]       {\( \alpha_{B, C, A} \)}           (P4);
  \end{tikzpicture}
