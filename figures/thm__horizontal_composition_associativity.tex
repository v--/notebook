\documentclass{classes/tikzcd}

\begin{document}
  \begin{tikzcd}[column sep=huge]
    \cat{A} \ar[r, bend left, "F" {description, name=F}] \ar[r, bend right, "G" {description, name=G}] & \cat{B} \ar[r, bend left, "H" {description, name=H}] \ar[r, bend right, "I" {description, name=I}] & \cat{C} \ar[r, bend left, "J" {description, name=J}] \ar[r, bend right, "K" {description, name=K}] & \cat{D}
    \ar[NatTrans, from=F, to=G, "\alpha"]
    \ar[NatTrans, from=H, to=I, "\beta"]
    \ar[NatTrans, from=J, to=K, "\gamma"]
  \end{tikzcd}

  \begin{tikzcd}[column sep=huge]
    \cat{A} \ar[r, bend left, "F" {description, name=F}] \ar[r, bend right, "G" {description, name=G}] & \cat{B} \ar[r, bend left, "{J \bincirc H}" {description, name=JH}] \ar[r, bend right, "{K \bincirc I}" {description, name=KI}] & \cat{D}
    \ar[NatTrans, from=F, to=G, "\alpha"]
    \ar[NatTrans, from=JH, to=KI, "\gamma \otimes \beta"]
  \end{tikzcd}

  \begin{tikzcd}[column sep=huge]
    \cat{A} \ar[r, bend left, "H \bincirc F" {description, name=HF}] \ar[r, bend right, "I \bincirc G" {description, name=IG}] & \cat{C} \ar[r, bend left, "J" {description, name=J}] \ar[r, bend right, "K" {description, name=K}] & \cat{D}
    \ar[NatTrans, from=HF, to=IG, "\beta \otimes \alpha"]
    \ar[NatTrans, from=J, to=K, "\gamma"]
  \end{tikzcd}

  \begin{tikzcd}[column sep=huge]
    {[H \bincirc F](A)} \ar[d, "\beta_{F(A)}" swap] \ar[r, "H(\alpha_A)"] & {[H \bincirc G](A)} \ar[d, "\beta_{G(A)}"] \\
    {[I \bincirc F](A)} \ar[r, dashed, "I(\alpha_A)" swap]                & {[I \bincirc G](A)}
  \end{tikzcd}

  \begin{tikzcd}[column sep=huge]
    {[J \bincirc H \bincirc F](A)} \ar[r, "{J(\beta_{F(A)})}"] & {[J \bincirc I \bincirc F](A)} \ar[d, dotted, "\gamma_{[I \bincirc F](A)}" swap] \ar[r, dashed, "{[J \bincirc I](\alpha_A)}"] & {[J \bincirc I \bincirc G](A)} \ar[d, "\gamma_{[I \bincirc G](A)}"] \\
                                                               & {[K \bincirc I \bincirc F](A)} \ar[r, "{[K \bincirc I]}(\alpha_A)" swap]                                                      & {[K \bincirc I \bincirc G](A)}
  \end{tikzcd}

  \begin{tikzcd}[column sep=huge]
    {[J \bincirc H](X)} \ar[d, "\gamma_{H(X)}" swap] \ar[r, "J(\beta_X)"] & {[J \bincirc I](X)} \ar[d, dotted, "\gamma_{I(X)}"] \\
    {[K \bincirc H](X)} \ar[r, "K(\beta_X)" swap]                         & {[K \bincirc I](X)}
  \end{tikzcd}
\end{document}
